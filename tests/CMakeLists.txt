cmake_minimum_required(VERSION 3.22)
project(framework_host_tests LANGUAGES C CXX)
set(CMAKE_C_STANDARD 11)
set(CMAKE_C_STANDARD_REQUIRED ON)
set(CMAKE_C_EXTENSIONS OFF)
set(CMAKE_CXX_STANDARD 23)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
enable_testing()
add_executable(remote_protocol_test remote_protocol_test.cpp
    ../Module/remote/Remote_receiver.cpp)
target_include_directories(remote_protocol_test PRIVATE ../Module/remote)
add_test(NAME remote_protocol COMMAND remote_protocol_test)

add_executable(protocol_test protocol_test.cpp)
target_include_directories(protocol_test PRIVATE ../Algorithm ../Module/transfer_protocol)
add_test(NAME packet_protocol COMMAND protocol_test)
set(USB_ROOT ../Middlewares/ST/STM32_USB_Device_Library)
add_executable(usb_test usb_test.cpp ../BSP/UsbPort.cpp
  ../USB_DEVICE/App/usbd_cdc_if.c ${USB_ROOT}/Class/CDC/Src/usbd_cdc.c)
target_include_directories(usb_test PRIVATE support ../BSP ../Algorithm
  ../USB_DEVICE/App ../USB_DEVICE/Target ${USB_ROOT}/Core/Inc ${USB_ROOT}/Class/CDC/Inc)
target_compile_options(usb_test PRIVATE -ffunction-sections -fdata-sections)
target_link_options(usb_test PRIVATE -Wl,--gc-sections)
add_test(NAME usb_backend COMMAND usb_test)
foreach(t remote_protocol_test protocol_test usb_test)
  target_compile_options(${t} PRIVATE -Wall -Wextra $<$<COMPILE_LANGUAGE:CXX>:-fno-exceptions;-fno-rtti>)
endforeach()
